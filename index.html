function login(){ const role = document.getElementById("role").value; const username = document.getElementById("username").value.trim(); const password = document.getElementById("password").value; const msg = document.getElementById("loginMessage"); msg.textContent = ""; if(username === "" || password === ""){ msg.textContent = "Username dan password wajib diisi!"; return; } /* ========================= LOGIN ADMIN ========================= */ if(role === "admin"){ if(username === "admin" && password === "12345"){ localStorage.setItem("loggedUser", "admin"); localStorage.setItem("loggedRole", "admin"); window.location.href = "admin.html"; }else{ msg.textContent = "Username atau password admin salah!"; } return; } /* ========================= LOGIN CUSTOMER ========================= */ let customers = JSON.parse( localStorage.getItem("customers") || "[]" ); let customer = customers.find( function(c){ return c.username === username; } ); /* CUSTOMER BARU */ if(!customer){ customer = { username: username, password: password }; customers.push(customer); localStorage.setItem( "customers", JSON.stringify(customers) ); } /* CUSTOMER LAMA */ else{ if(customer.password !== password){ msg.textContent = "Password customer salah!"; return; } } /* SIMPAN SESSION */ localStorage.setItem( "loggedUser", username ); localStorage.setItem( "loggedRole", "customer" ); /* MASUK KE APLIKASI */ showApp(); }
